Line plot for height over time

height_overtime_plot <- height_data_clean %>%
  ggplot(aes(x = survey_date, y = height_cm, 
             group = pop, color = pop)) +
  geom_smooth(method = "lm", se = F) +
  labs(title = "Height Over Time", x = "Survey Date", y = "Height (cm)") +
  theme_minimal()

Box plot for Jan 27, 2023

height_Jan27_plot <- height_data_clean %>%
  filter(survey_date == '2023-01-27') %>%
  ggplot(aes(x = pop, y = height_cm, group = pop, fill = pop)) +
  geom_boxplot() +
  labs(title = "Height Distribution on Jan 27, 2023", x = "Population", y = "Height (cm)") +
  theme_minimal()
